/* =========================================================
   HEADER SCROLL EFFECT
========================================================= */

function updateHeader() {

    if (!header) {
        return;
    }

    if (window.scrollY > 20) {

        header.classList.add("scrolled");

    } else {

        header.classList.remove("scrolled");

    }

}

window.addEventListener(
    "scroll",
    updateHeader,
    { passive: true }
);

updateHeader();


/* =========================================================
   MOBILE NAVIGATION
========================================================= */

function closeMobileMenu() {

    if (!mobileMenuBtn || !mobileNav) {
        return;
    }

    mobileMenuBtn.classList.remove("open");

    mobileNav.classList.remove("open");

    mobileMenuBtn.setAttribute(
        "aria-expanded",
        "false"
    );

}


if (mobileMenuBtn && mobileNav) {

    mobileMenuBtn.addEventListener(
        "click",
        () => {

            const isOpen =
                mobileNav.classList.toggle("open");

            mobileMenuBtn.classList.toggle(
                "open",
                isOpen
            );

            mobileMenuBtn.setAttribute(
                "aria-expanded",
                String(isOpen)
            );

        }
    );

}


navLinks.forEach((link) => {

    link.addEventListener(
        "click",

            closeMobileMenu();

        }
    );

});

    "keydown",
    (event) => {

        if (event.key === "Escape") {

            closeMobileMenu();

            closeReviewModalHandler();

        }

    }
);


/* =========================================================
   ACTIVE NAVIGATION
========================================================= */

const observerOptions = {
    root: null,
    rootMargin: "-25% 0px -60% 0px",
    threshold: 0
};


const sectionObserver =
    new IntersectionObserver(
        (entries) => {

            entries.forEach((entry) => {

                if (!entry.isIntersecting) {
                    return;
                }

                const currentId =
                    entry.target.getAttribute("id");

                document
                    .querySelectorAll(
                        ".desktop-nav .nav-link"
                    )
                    .forEach((link) => {

                        const href =
                            link.getAttribute("href");

                        link.classList.toggle(
                            "active",
                            href === `#${currentId}`
                        );

                    });

            });

        },
        observerOptions
    );


sections.forEach((section) => {

    sectionObserver.observe(section);

});


/* =========================================================
   SCROLL REVEAL
========================================================= */

const revealObserver =
    new IntersectionObserver(
        (entries, observer) => {

            entries.forEach((entry) => {

                if (!entry.isIntersecting) {
                    return;
                }

                entry.target.classList.add("visible");

                observer.unobserve(entry.target);

            });

        },
        {
            threshold: 0.12,
            rootMargin: "0px 0px -40px 0px"
        }
    );


revealElements.forEach((element) => {

    revealObserver.observe(element);

});


/* =========================================================
   NUMBER COUNTER
========================================================= */

let countersStarted = false;


function animateCounter(element) {

    const target =
        Number(element.dataset.count);

    if (
        !Number.isFinite(target) ||
        target < 0
    ) {
        return;
    }

    const duration = 1600;

    const startTime = performance.now();


    function updateCounter(currentTime) {

        const elapsed =
            currentTime - startTime;

        const progress =
            Math.min(elapsed / duration, 1);

        const eased =
            1 - Math.pow(1 - progress, 3);

        const currentValue =
            Math.floor(target * eased);

        element.textContent =
            currentValue.toLocaleString();

        if (progress < 1) {

            requestAnimationFrame(
                updateCounter
            );

        } else {

            element.textContent =
                target.toLocaleString();

        }

    }


    requestAnimationFrame(updateCounter);

}


const statsSection =
    document.querySelector(".stats-section");


if (statsSection) {

    const statsObserver =
        new IntersectionObserver(
            (entries, observer) => {

                if (
                    entries[0].isIntersecting &&
                    !countersStarted
                ) {

                    countersStarted = true;

                    counterElements.forEach(
                        animateCounter
                    );

                    observer.disconnect();

                }

            },
            {
                threshold: 0.35
            }
        );


    statsObserver.observe(statsSection);

}


/* =========================================================
   ENQUIRY FORM VALIDATION
========================================================= */

function setFieldError(
    field,
    message
) {

    const group =
        field.closest(".form-group");

    if (!group) {
        return;
    }

    group.classList.toggle(
        "error",
        Boolean(message)
    );

    const error =
        group.querySelector(".form-error");

    if (error) {

        error.textContent =
            message || "";

    }

}


function validateRequiredField(
    field,
    message
) {

    if (!field.value.trim()) {

        setFieldError(
            field,
            message
        );

        return false;

    }

    setFieldError(field, "");

    return true;

}


function validatePhone(field) {

    const value =
        field.value.replace(/\D/g, "");

    if (!value) {

        setFieldError(
            field,
            "Mobile number is required."
        );

        return false;

    }

    if (value.length < 10) {

        setFieldError(
            field,
            "Enter a valid mobile number."
        );

        return false;

    }

    setFieldError(field, "");

    return true;

}


if (enquiryForm) {

    enquiryForm.addEventListener(
        "submit",
        (event) => {

            event.preventDefault();

            const parentName =
                document.getElementById(
                    "parentName"
                );

            const phone =
                document.getElementById(
                    "phone"
                );

            const studentClass =
                document.getElementById(
                    "studentClass"
                );

            const subject =
                document.getElementById(
                    "subject"
                );

            const validName =
                validateRequiredField(
                    parentName,
                    "Please enter your name."
                );

            const validPhone =
                validatePhone(phone);

            const validClass =
                validateRequiredField(
                    studentClass,
                    "Please select a class."
                );

            const validSubject =
                validateRequiredField(
                    subject,
                    "Please select a subject."
                );


            if (
                !validName ||
                !validPhone ||
                !validClass ||
                !validSubject
            ) {

                const firstError =
                    enquiryForm.querySelector(
                        ".form-group.error input, .form-group.error select"
                    );

                if (firstError) {
                    firstError.focus();
                }

                return;

            }


            /*
             * FRONTEND ONLY:
             *
             * This is intentionally not pretending
             * to send data to a real server.
             *
             * When the backend is connected, replace
             * this section with a secure POST request
             * to the enquiry API.
             */


            enquiryForm.reset();

            showToast(
                "Your enquiry has been submitted successfully."
            );

        }
    );


    const liveFields =
        enquiryForm.querySelectorAll(
            "input, select, textarea"
        );


    liveFields.forEach((field) => {

        field.addEventListener(
            "input",
            () => {

                if (
                    field.closest(".form-group")
                        ?.classList.contains("error")
                ) {

                    setFieldError(
                        field,
                        ""
                    );

                }

            }
        );


        field.addEventListener(
            "change",
            () => {

                if (
                    field.closest(".form-group")
                        ?.classList.contains("error")
                ) {

                    setFieldError(
                        field,
                        ""
                    );

                }

            }
        );

    });

}


/* =========================================================
   REVIEW MODAL
========================================================= */

function openReviewModalHandler() {

    if (!reviewModal) {
        return;
    }

    reviewModal.classList.add("active");

    reviewModal.setAttribute(
        "aria-hidden",
        "false"
    );

    document.body.style.overflow = "hidden";

    setTimeout(() => {

        const reviewName =
            document.getElementById(
                "reviewName"
            );

        if (reviewName) {
            reviewName.focus();
        }

    }, 100);

}


function closeReviewModalHandler() {

    if (!reviewModal) {
        return;
    }

    reviewModal.classList.remove("active");

    reviewModal.setAttribute(
        "aria-hidden",
        "true"
    );

    document.body.style.overflow = "";

}


if (openReviewModal) {

    openReviewModal.addEventListener(
        "click",
        openReviewModalHandler
    );

}


if (closeReviewModal) {

    closeReviewModal.addEventListener(
        "click",
        closeReviewModalHandler
    );

}


if (modalOverlay) {

    modalOverlay.addEventListener(
        "click",
        closeReviewModalHandler
    );

}


/* =========================================================
   STAR RATING
========================================================= */

let selectedRating = 0;


function updateRatingUI() {

    ratingButtons.forEach(
        (button) => {

            const rating =
                Number(
                    button.dataset.rating
                );

            button.classList.toggle(
                "selected",
                rating <= selectedRating
            );

        }
    );

}


ratingButtons.forEach(
    (button) => {

        button.addEventListener(
            "click",
            () => {

                selectedRating =
                    Number(
                        button.dataset.rating
                    );

                updateRatingUI();

            }
        );

    }
);


/* =========================================================
   REVIEW FORM
========================================================= */

if (reviewForm) {

    reviewForm.addEventListener(
        "submit",
        (event) => {

            event.preventDefault();

            const name =
                document.getElementById(
                    "reviewName"
                );

            const studentClass =
                document.getElementById(
                    "reviewClass"
                );

            const message =
                document.getElementById(
                    "reviewMessage"
                );


            if (!name.value.trim()) {

                name.focus();

                showToast(
                    "Please enter your name."
                );

                return;

            }


            if (!studentClass.value) {

                studentClass.focus();

                showToast(
                    "Please select the student class."
                );

                return;

            }


            if (selectedRating === 0) {

                showToast(
                    "Please select a rating."
                );

                return;

            }


            if (!message.value.trim()) {

                message.focus();

                showToast(
                    "Please write your review."
                );

                return;

            }


            /*
             * FRONTEND DEMONSTRATION ONLY:
             *
             * The review UI is ready.
             *
             * For the real production version,
             * this form should POST to a backend/API.
             *
             * The backend can store reviews in a database
             * and return approved/published reviews.
             */


            reviewForm.reset();

            selectedRating = 0;

            updateRatingUI();

            closeReviewModalHandler();

            showToast(
                "Thank you for sharing your experience!"
            );

        }
    );

}


/* =========================================================
   TOAST
========================================================= */

let toastTimer;


function showToast(message) {

    if (!toast || !toastMessage) {
        return;
    }

    toastMessage.textContent = message;

    toast.classList.add("show");

    clearTimeout(toastTimer);

    toastTimer = setTimeout(
        () => {

            toast.classList.remove("show");

        },
        3500
    );

}


/* =========================================================
   SMOOTH ANCHOR BEHAVIOUR
========================================================= */

document
    .querySelectorAll(
        'a[href^="#"]'
    )
    .forEach((link) => {

        link.addEventListener(
            "click",
            (event) => {

                const targetId =
                    link.getAttribute("href");

                if (
                    !targetId ||
                    targetId === "#"
                ) {
                    return;
                }

                const target =
                    document.querySelector(
                        targetId
                    );

                if (!target) {
                    return;
                }

                event.preventDefault();

                target.scrollIntoView({
                    behavior: "smooth",
                    block: "start"
                });

            }
        );

    });
